```python
import smac
from smac import sc
from sklearn.datasets import load_iris
from sklearn.model_selection import train_test_split

# Load dataset
iris = load_iris()
X_train, X_test, y_train, y_test = train_test_split(iris.data, iris.target, test_size=0.2)

# Define configuration space
config_space = {
    'n_estimators': [10, 50, 100],
    'max_depth': [None, 5, 10],
}

# Create SMAC instance
smac_instance = smac.SMAC(
    config_space=config_space,
    target_function=sc.accuracy,
)

# Run SMAC
smac_instance.run(X_train, y_train)

# Get best configuration
best_config = smac_instance.get_best_configuration()

# Evaluate best configuration
y_pred = best_config.predict(X_test)
```
